4.  func shortTotal(_ Commit: [Int]) -> (filters: [Int], sorted: [Int], total: Int) {
    // A. hapus nilai 0
    let filtered = Commit.filter{ $0 != 0}
    
    //B. shorting dari terbersar ke kecil
    let shorted = filtered.sorted(by: >)
    
    //C. Jumlah total
    let total = filtered.reduce(0, +)
    
    return (filters, shorted, total)
}

5. func validateUser(_ user: [String: String?]) -> String {
    guard let nameOptional = user["name"],
          let name = nameOptional,
          !name.isEmpty
    else { return "Error: Name is Missing" }
    
    guard let emailOptional = user["email"],
          let email = emailOptional,
          !email.isEmpty
    else { return "Error: Email is Missing" }
    
    let role = user["role"] ?? "role unknown"
    
    return "User \(name) dengan role \(role) valid"
}

disini saya membuat guard untuk validasi name dan email, dan juga untuk mendefinisikan string object dari user. jika salah satu name atau email tidak ada akan muncul error message dari return masing masing guard. untuk role saya membuat optional berisi role uknown jika tidak ada. karena tidak diminta validasi untuk role.
